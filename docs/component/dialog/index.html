use crate::components::button::{Button, ButtonSize, ButtonVariant};
use dioxus::prelude::*;
use dioxus_icons::lucide::X;
use dioxus_primitives::dialog::{self, DialogCtx, DialogDescriptionProps, DialogTitleProps};
use dioxus_primitives::keyboard_inset::{use_keyboard_panel, KeyboardInset};
use dioxus_primitives::{dioxus_attributes::attributes, merge_attributes};

/// The props for the [`Dialog`] component: [`dialog::DialogRootProps`] plus `show_close_button`.
#[derive(Props, Clone, PartialEq)]
pub struct DialogProps {
    /// The ID of the dialog root element.
    pub id: ReadSignal<Option<String>>,

    /// Whether the dialog is modal. If true, it traps focus within the dialog when open.
    #[props(default = ReadSignal::new(Signal::new(true)))]
    pub is_modal: ReadSignal<bool>,

    /// Whether a modal dialog dims the page behind it with the shared overlay scrim.
    /// Defaults to `true`; see [`dialog::DialogRootProps::overlay`].
    #[props(default = ReadSignal::new(Signal::new(true)))]
    pub overlay: ReadSignal<bool>,

    /// Base UI's `disablePointerDismissal`; see [`dialog::DialogRootProps::disable_pointer_dismissal`].
    #[props(default = ReadSignal::new(Signal::new(false)))]
    pub disable_pointer_dismissal: ReadSignal<bool>,

    /// The controlled `open` state of the dialog.
    pub open: ReadSignal<Option<bool>>,

    /// The default `open` state of the dialog if it is not controlled.
    #[props(default)]
    pub default_open: bool,

    /// A callback that is called when the open state changes.
    #[props(default)]
    pub on_open_change: Callback<bool>,

    /// Whether the panel renders its own close button, an `X` at the top end. Defaults to `true`,
    /// as shadcn's `DialogContent showCloseButton`. The button is the last child of the panel, so
    /// a keyboard open focuses the first control in `children` (the first field), not the X.
    #[props(default = true)]
    pub show_close_button: bool,

    /// Where focus lands on open, as a function of the open modality
    /// ([`dioxus_primitives::modality::OpenModality`]); see [`dialog::DialogContentProps::initial_focus`].
    #[props(default)]
    pub initial_focus: Option<dioxus_primitives::modality::FocusTargetFn>,

    /// Where focus goes on close; see [`dialog::DialogContentProps::final_focus`].
    #[props(default)]
    pub final_focus: Option<dioxus_primitives::modality::FocusTargetFn>,

    /// Additional attributes to apply to the dialog panel element.
    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,

    /// The children of the dialog.
    pub children: Element,
}

#[component]
pub fn Dialog(props: DialogProps) -> Element {
    let base = attributes!(div {
        class: "dx-dialog",
        "data-slot": "dialog-content",
    });
    let merged = merge_attributes(vec![base, props.attributes]);
    // The panel follows the on-screen keyboard (`primitives/src/keyboard_inset.rs`; `style.css`
    // consumes `--dx-keyboard-inset`).
    let panel = use_keyboard_panel(None);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/dialog/style.css") }
        dialog::DialogRoot {
            class: "dx-dialog-backdrop",
            id: props.id,
            is_modal: props.is_modal,
            overlay: props.overlay,
            disable_pointer_dismissal: props.disable_pointer_dismissal,
            open: props.open,
            default_open: props.default_open,
            on_open_change: props.on_open_change,
            dialog::DialogContent {
                id: panel.id(),
                class: None,
                initial_focus: props.initial_focus,
                final_focus: props.final_focus,
                attributes: merged,
                {props.children}
                if props.show_close_button {
                    DialogCornerClose {}
                }
                KeyboardInset { panel }
            }
        }
    }
}

/// The panel's own close button (`show_close_button`). Its own function so nothing the panel
/// merges onto `.dx-dialog` is in scope here: the Button receives only the close behaviour.
///
/// shadcn's `DialogContent` renders the X after `children`: it is last in DOM order, so
/// `showModal()`'s initial focus (the first focusable descendant) lands on the first control of
/// the body. The wrapper owns the placement; the button is the real Button, untouched by this
/// sheet.
#[component]
fn DialogCornerClose() -> Element {
    rsx! {
        div { class: "dx-dialog-close", "data-slot": "dialog-close-slot",
            DialogClose {
                "data-slot": "dialog-close",
                aria_label: "Close",
                r#as: |close_attributes| rsx! {
                    Button {
                        variant: ButtonVariant::Ghost,
                        size: ButtonSize::IconSm,
                        attributes: close_attributes,
                        X {}
                    }
                },
            }
        }
    }
}

/// Closes the enclosing [`Dialog`] (shadcn's `DialogClose`). Renders a `<button type="button">`;
/// `r#as` renders another element (typically a `Button`) with the close behaviour in its
/// `attributes`, as shadcn's `render={<Button />}`.
// No `document::Link` (as `SheetClose`): it reads `DialogCtx`, so it only renders inside a
// `Dialog`, which already links the stylesheet.
#[component]
pub fn DialogClose(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    r#as: Option<Callback<Vec<Attribute>, Element>>,
    children: Element,
) -> Element {
    let ctx: DialogCtx = use_context();

    let base = attributes!(button {
        r#type: "button",
        onclick: move |_| {
            ctx.set_open(false);
        }
    });
    let merged = merge_attributes(vec![base, attributes]);

    if let Some(dynamic) = r#as {
        dynamic.call(merged)
    } else {
        rsx! {
            button { ..merged, {children} }
        }
    }
}

/// Groups the title and description (shadcn's `DialogHeader`: a column with `gap-2`).
#[component]
pub fn DialogHeader(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    children: Element,
) -> Element {
    let base = attributes!(div { class: "dx-dialog-header", "data-slot": "dialog-header" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/dialog/style.css") }
        div { ..merged, {children} }
    }
}

/// The action row (shadcn's `DialogFooter`): a muted band across the panel's foot, a column
/// reversed on a narrow screen and a row aligned to the end from `sm` up. `show_close_button`
/// appends an outline "Close" button, as shadcn's.
#[component]
pub fn DialogFooter(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    #[props(default = false)] show_close_button: bool,
    children: Element,
) -> Element {
    let base = attributes!(div { class: "dx-dialog-footer", "data-slot": "dialog-footer" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/dialog/style.css") }
        div { ..merged,
            {children}
            if show_close_button {
                DialogClose {
                    r#as: |close_attributes| rsx! {
                        Button { variant: ButtonVariant::Outline, attributes: close_attributes, "Close" }
                    },
                }
            }
        }
    }
}

#[component]
pub fn DialogTitle(props: DialogTitleProps) -> Element {
    let base = attributes!(h2 {
        class: "dx-dialog-title",
    });
    let merged = merge_attributes(vec![base, props.attributes]);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/dialog/style.css") }
        dialog::DialogTitle {
            id: props.id,
            attributes: merged,
            {props.children}
        }
    }
}

#[component]
pub fn DialogDescription(props: DialogDescriptionProps) -> Element {
    let base = attributes!(p {
        class: "dx-dialog-description",
    });
    let merged = merge_attributes(vec![base, props.attributes]);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/dialog/style.css") }
        dialog::DialogDescription {
            id: props.id,
            attributes: merged,
            {props.children}
        }
    }
}

The dialog component can be used to display additional information or actions related to an item. It is a simple modal dialog that can be opened and closed by the user.

// The dialog component must wrap all dialog elements.
Dialog {
    // The open prop determines if the dialog is currently open or closed.
    open: open(),
    // Header: the title and description in a column (shadcn's DialogHeader).
    DialogHeader {
        DialogTitle { "Item information" }
        DialogDescription { "Here is some additional information about the item." }
    }
    // Footer: the action row, a muted band across the foot of the panel.
    DialogFooter {
        DialogClose {
            r#as: |attributes| rsx! { Button { variant: ButtonVariant::Outline, attributes, "Cancel" } },
        }
        Button { r#type: "submit", "Save changes" }
    }
}

Dialog renders its own close button, an X at the top end of the panel, as shadcn's DialogContent does (showCloseButton, default on). It is the real Button (ghost, icon-sm) named "Close", inside a wrapper that only places it. Pass show_close_button: false to leave it out (Escape and a click on the scrim still dismiss the dialog), as in the "No close button" example.

The button is the last child of the panel, in shadcn's DOM order. A dialog opened from the keyboard focuses its first control (see use_dialog_open_driver), so a dialog with a form lands on the first field and never on the X; a dialog with no other control lands on the X. A pointer or touch open focuses the panel itself.

DialogFooter { show_close_button: true } appends an outline "Close" button to the footer, as shadcn's. DialogClose is the close behaviour on its own, shadcn's DialogClose: a <button type="button"> that closes the enclosing dialog, or any element through r#as (the attributes carry the click handler), the way SheetClose takes as.

DialogHeader is a column with a 0.5rem gap. DialogFooter cancels the panel's padding (-mx-4 -mb-4) to draw a muted/50 band with a top border and the panel's bottom corners; it is a reversed column below sm (the primary action on top) and a row aligned to the end from sm up. A title and description placed directly in the panel (no header) keep the tighter pull-back of the previous release.

A modal dialog dims the page behind it with the shared overlay scrim: the dialog's own ::backdrop, the same 10% black and 4px blur on every modal overlay. Pass overlay: false to keep the modality (focus trap, inert page, click-outside dismissal) without the dim and the blur; the <dialog> then carries data-dx-overlay="off".

Closing keeps the dialog a modal -- in the top layer, at the same size and place -- until its exit animation has played, and only then closes it. That holds in every browser; no stylesheet needs an engine-specific keep-alive.

A dialog is not closed the instant open turns false. It stays open (in the top layer, at the same size and place) until its own exit animation and its backdrop's have settled, at most 1500 ms, and only then calls close(), so any exit animation you write on the dialog or its ::backdrop is played in full. The scrim and the panel share --dx-overlay-duration and --dx-overlay-ease (defaults --dx-motion-duration-slow and --dx-motion-ease); retune them in one place, or pass overlay: false for no scrim. Keep exit keyframes to transform, translate and opacity. Under prefers-reduced-motion: reduce both durations become --dx-motion-duration-reduced. Nothing here loops, so there is nothing to pause while unseen. See Motion for the tokens, reduced motion and the per-component table.

initial_focus and final_focus take a function of the open modality (OpenModality::{Keyboard, Mouse, Touch, Pen}, Base UI's initialFocus(openType)) that returns a FocusTarget: Default, Panel, Selector("#id") or, for final_focus, None. They are inputs to the one open-focus policy (dxOpenFocus / dxCloseFocus, primitives/src/modality.rs), not a second code path: the function is called once per modality at render, the answers ride on the panel as data-dx-initial-focus / data-dx-final-focus, and the policy reads the entry for the modality of the open. A selector that matches nothing falls back to the built-in policy (keyboard and mouse: the first control; touch and pen: the panel, so no software keyboard rises). An explicit target is honoured as given, so return Panel for Touch yourself when it names a field.

Dialog {
    initial_focus: Callback::new(|m: OpenModality| match m {
        OpenModality::Touch => FocusTarget::Panel,
        _ => FocusTarget::selector("#name"),
    }),
    final_focus: Callback::new(|_| FocusTarget::Default),
}

final_focus defaults to the element that held focus when the panel opened: the trigger for a click open, the previously focused element for a programmatic open. The opener is remembered in exactly one place (dxOpenFocus), for every overlay kind. See the focus_target variant.

is_modal: false renders a plain panel beside the page (no top layer, no trap, no inert page). It reports role="dialog" but no aria-modal: the attribute is derived from the arm in one helper, so only a modal panel ever tells assistive tech the page behind it is inert. By default a press or focus outside the panel dismisses it. disable_pointer_dismissal: true (Base UI's disablePointerDismissal) turns that off: an outside press only acts on the page (on a modal, a backdrop click no longer closes it). Escape and the dialog's own close controls still close it.
