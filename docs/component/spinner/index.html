use dioxus::prelude::*;
use crate::components::icons::LoaderCircle;
use dioxus_primitives::activity::use_motion;
use dioxus_primitives::dioxus_attributes::attributes;
use dioxus_primitives::merge_attributes;

/// The ring's size, shadcn's `size-*` utilities as a type: 12, 16 (default), 24 or 32 px.
#[derive(Copy, Clone, PartialEq, Eq, Default, Debug)]
#[non_exhaustive]
pub enum SpinnerSize {
    Sm,
    #[default]
    Default,
    Lg,
    Xl,
}

impl SpinnerSize {
    /// The `data-size` value the stylesheet keys on.
    pub fn class(&self) -> &'static str {
        match self {
            SpinnerSize::Sm => "sm",
            SpinnerSize::Default => "default",
            SpinnerSize::Lg => "lg",
            SpinnerSize::Xl => "xl",
        }
    }
}

/// A loading indicator: a continuously rotating ring icon. Renders
/// `role="status"` with an accessible name (default `"Loading"`, overridable
/// via `label`) so assistive tech announces the busy state -- there is no
/// ARIA widget pattern beyond that live-region role, so no primitive
/// underneath.
#[component]
pub fn Spinner(
    /// The accessible name announced for the loading state.
    #[props(default = "Loading".to_string())]
    label: String,
    /// The ring's size. Defaults to 16px.
    #[props(default)]
    size: SpinnerSize,
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
) -> Element {
    let base = attributes!(span { class: "dx-spinner", "data-dx-root": "", "aria-label": "{label}" });
    // `role="status"` is required live-region semantics, not a caller
    // default -- owned-wins.
    let owned = attributes!(span { role: "status", "data-size": size.class() });
    // Paused (not stopped) while off-screen or in a hidden tab: `data-dx-motion="paused"` is turned
    // into `animation-play-state: paused` by `dx-components-theme.css`, so the ring keeps its
    // phase and costs nothing while nobody can see it.
    let motion = use_motion();
    let merged = merge_attributes(vec![base, attributes, owned, motion.attributes()]);
    // `dx-size-own` is shadcn's `size-*` class on the spinner's svg: a container's icon rule
    // (`.dx-button svg:not([class*="size-"])` and its kin) skips an svg that sizes itself, so the ring
    // keeps its own edge inside a Button of any size (backlog row 188).
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/spinner/style.css") }
        span { ..merged, LoaderCircle { class: "dx-size-own" } }
    }
}

Spinner {
    // The accessible name announced for the loading state. Defaults to "Loading".
    label: "Saving",
}

Spinner Styled component

A loading indicator: a continuously rotating ring icon. Renders role="status" with an accessible name (default "Loading", overridable via label) so assistive tech announces the busy state -- there is no ARIA widget pattern beyond that live-region role, so no primitive underneath.

PropTypeDefaultDescription
labelString"Loading".to_string()

The accessible name announced for the loading state.

sizeSpinnerSizeDefault::default()

The ring's size. Defaults to 16px.

Plus the global attributes.

Component-Specific

Spinner { size } takes SpinnerSize::Sm (12px), Default (16px), Lg (24px) and Xl (32px). The spinner carries data-size; the ring's edge is the one slot variable --dx-spinner-size, which a parent can also set.

The ring spins forever, so it pauses whenever nobody can see it: while it is outside the viewport (plus a 200px pre-roll, so it is already turning when it scrolls in), inside a skipped content-visibility subtree, or in a hidden tab. Paused is animation-play-state: paused, not removed, so it keeps its angle and resumes seamlessly at zero cost.

This is built in. Spinner calls dioxus_primitives::activity::use_motion() and spreads motion.attributes() on its root; the one [data-dx-motion="paused"] rule in dx-components-theme.css does the pausing for the element and everything under it. A page of spinners shares one IntersectionObserver. If you build your own looping indicator, do the same two lines rather than writing a pause rule. Users who prefer reduced motion are handled by your stylesheet's prefers-reduced-motion query, not by this.

The spin is transform: rotate, 1 s linear; override animation-duration on .dx-spinner svg to change it. See Motion for the tokens, reduced motion and the per-component table.

A component that composes this one retunes it by setting one of these custom properties on its own element, never by declaring the property on its own class: two stylesheets declaring one property on one element are decided by which sheet loads last, and that order differs from route to route. The component reads each variable with its own default, so a variable that is not set changes nothing. Custom properties inherit: set one on the exact element, not on an ancestor, or it reaches every nested instance too.

VariableDefaultPropertyDescription
--dx-spinner-size1rem--dx-spinner-edge

Edge length of a spinner nested in a container that sizes its icons (a Badge sets 0.75rem). The container sets it on its own element; a spinner outside one is 1rem.
