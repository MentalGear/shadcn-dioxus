use dioxus::prelude::*;
use dioxus_primitives::dioxus_attributes::attributes;
use dioxus_primitives::avatar::{self, AvatarState};
use dioxus_primitives::merge_attributes;

/// The avatar's size, following shadcn's Nova style (`data-[size=sm]:size-6`, the
/// default `size-8`, `data-[size=lg]:size-10`).
///
/// The variant names keep the library's own spelling; the sizes are shadcn's scale:
///
/// | variant      | size           | shadcn `size` |
/// |--------------|----------------|---------------|
/// | `ExtraSmall` | 24px           | `sm`          |
/// | `Small`      | 32px (default) | `default`     |
/// | `Medium`     | 40px           | `lg`          |
#[derive(Clone, Copy, PartialEq, Default)]
pub enum AvatarImageSize {
    ExtraSmall,
    #[default]
    Small,
    Medium,
}

impl AvatarImageSize {
    /// The value of `data-size` on the root, shadcn's own spelling.
    fn to_data_size(self) -> &'static str {
        match self {
            AvatarImageSize::ExtraSmall => "sm",
            AvatarImageSize::Small => "default",
            AvatarImageSize::Medium => "lg",
        }
    }
}

#[derive(Clone, Copy, PartialEq, Default)]
pub enum AvatarShape {
    #[default]
    Circle,
    Rounded,
}

impl AvatarShape {
    fn to_class(self) -> &'static str {
        match self {
            AvatarShape::Circle => "dx-avatar-circle",
            AvatarShape::Rounded => "dx-avatar-rounded",
        }
    }
}

/// The props for the [`Avatar`] root component.
#[derive(Props, Clone, PartialEq)]
pub struct AvatarProps {
    /// Callback when image loads successfully.
    #[props(default)]
    pub on_load: Option<EventHandler<()>>,

    /// Callback when image fails to load.
    #[props(default)]
    pub on_error: Option<EventHandler<()>>,

    /// Callback when the avatar state changes.
    #[props(default)]
    pub on_state_change: Option<EventHandler<AvatarState>>,

    /// The avatar's size: `ExtraSmall`, `Small` (default) or `Medium`.
    #[props(default)]
    pub size: AvatarImageSize,

    /// The avatar's outline: see `AvatarShape` (default `Circle`).
    #[props(default)]
    pub shape: AvatarShape,

    /// Additional attributes for the avatar element.
    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,

    /// The fallback content shown while the image is loading or if it fails to load.
    pub children: Element,
}

#[component]
pub fn Avatar(props: AvatarProps) -> Element {
    let class = format!("{} {}", "dx-avatar", props.shape.to_class());
    let base = attributes!(span {
        class,
        "data-dx-root": "",
        "data-size": props.size.to_data_size(),
    });
    let merged = merge_attributes(vec![base, props.attributes]);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/avatar/style.css") }
        avatar::Avatar {
            on_load: props.on_load,
            on_error: props.on_error,
            on_state_change: props.on_state_change,
            attributes: merged,
            {props.children}
        }
    }
}

#[derive(Props, Clone, PartialEq)]
pub struct AvatarImageProps {
    /// The `id` of the `<img>` element.
    #[props(default)]
    pub id: ReadSignal<Option<String>>,

    /// The image URL. While it loads (or if it fails) the sibling `AvatarFallback` shows.
    pub src: String,

    /// Alternative text for the image; leave empty only for a purely decorative avatar.
    #[props(default)]
    pub alt: String,

    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,
}

#[component]
pub fn AvatarImage(props: AvatarImageProps) -> Element {
    let base = attributes!(img {
        class: "dx-avatar-image",
        draggable: "false",
    });
    let merged = merge_attributes(vec![base, props.attributes]);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/avatar/style.css") }
        avatar::AvatarImage {
            id: props.id,
            src: props.src,
            alt: props.alt,
            attributes: merged,
        }
    }
}

#[derive(Props, Clone, PartialEq)]
pub struct AvatarFallbackProps {
    /// Milliseconds to wait before the fallback may show (Base UI's `delay`): a picture that loads within it never
    /// flashes the fallback. Default: none, the fallback shows immediately (also in the prerendered page).
    #[props(default)]
    pub delay: Option<u32>,

    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,

    /// What shows while the image loads or after it fails: initials, an icon.
    pub children: Element,
}

#[component]
pub fn AvatarFallback(props: AvatarFallbackProps) -> Element {
    let base = attributes!(span {
        class: "dx-avatar-fallback",
    });
    let merged = merge_attributes(vec![base, props.attributes]);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/avatar/style.css") }
        avatar::AvatarFallback {
            delay: props.delay,
            attributes: merged,
            {props.children}
        }
    }
}

/// The props for the [`ImageAvatar`] convenience component.
#[derive(Props, Clone, PartialEq)]
pub struct ImageAvatarProps {
    /// The image source URL.
    pub src: String,

    /// The image alt text.
    #[props(default)]
    pub alt: String,

    /// Callback when image loads successfully.
    #[props(default)]
    pub on_load: Option<EventHandler<()>>,

    /// Callback when image fails to load.
    #[props(default)]
    pub on_error: Option<EventHandler<()>>,

    /// Callback when the avatar state changes.
    #[props(default)]
    pub on_state_change: Option<EventHandler<AvatarState>>,

    /// The avatar's size: `ExtraSmall`, `Small` (default) or `Medium`.
    #[props(default)]
    pub size: AvatarImageSize,

    /// The avatar's outline: see `AvatarShape` (default `Circle`).
    #[props(default)]
    pub shape: AvatarShape,

    /// Additional attributes for the avatar element.
    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,

    /// Milliseconds to hold the fallback back while the image loads (see [`AvatarFallback`]'s `delay`).
    #[props(default)]
    pub fallback_delay: Option<u32>,

    /// The fallback content shown while the image is loading or if it fails to load.
    pub children: Element,
}

#[component]
pub fn ImageAvatar(props: ImageAvatarProps) -> Element {
    // axe `role-img-alt` (dev-docs/backlog.md row 34): the root `Avatar` renders
    // `role="img"` (`avatar.rs`) and gets its accessible name from a
    // caller-supplied `aria_label`, same as this crate's own doc example.
    // `alt` alone reaches only the inner `<img>` (via `AvatarImage`), never the
    // outer `role="img"` span, so a caller that passes `alt` without also
    // setting `aria_label` would leave an unnamed `role="img"`. The root's
    // `aria_label` therefore defaults from `alt` -- merged caller-wins, so an
    // explicit `aria_label` in `props.attributes` still overrides it. Contributed
    // only when `alt` is non-empty: an `aria-label=""` is still an empty
    // accessible name (no better than omitting it), and would additionally
    // shadow any real name the caller supplies some other way.
    let base: Vec<Attribute> = if props.alt.is_empty() {
        Vec::new()
    } else {
        attributes!(span { aria_label: "{props.alt}" })
    };
    let attributes = merge_attributes(vec![base, props.attributes]);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/avatar/style.css") }
        Avatar {
            on_load: props.on_load,
            on_error: props.on_error,
            on_state_change: props.on_state_change,
            size: props.size,
            shape: props.shape,
            attributes,
            AvatarImage {
                src: props.src,
                alt: props.alt,
            }
            AvatarFallback { delay: props.fallback_delay, {props.children} }
        }
    }
}

/// A status dot or icon at the avatar's bottom-end corner (shadcn's `AvatarBadge`: `absolute right-0
/// bottom-0`, `bg-primary text-primary-foreground`, a 2px ring in the page background). It must be a child
/// of [`Avatar`] -- the size comes from the root's `data-size`: 8px in the small avatar (an icon is hidden
/// there), 10px by default and 12px in the large one, with the icon at 8px. Colour a status by overriding
/// `background-color` (`style: "background-color: var(--dx-success)"`). A purely decorative mark; give the
/// avatar's `aria-label` the status if it matters.
#[component]
pub fn AvatarBadge(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    #[props(default)] children: Element,
) -> Element {
    let base = attributes!(span { class: "dx-avatar-badge" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/avatar/style.css") }
        span { ..merged, {children} }
    }
}

/// An overlapping stack of [`Avatar`]s (shadcn's `AvatarGroup`: `flex -space-x-2`, each avatar ringed in
/// the page background so the overlap reads as a cut). End it with an [`AvatarGroupCount`] for the "+N"
/// tile. A plain `div`; pass `aria_label` to name the group.
#[component]
pub fn AvatarGroup(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    children: Element,
) -> Element {
    let base = attributes!(div { class: "dx-avatar-group" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/avatar/style.css") }
        div { ..merged, {children} }
    }
}

/// The "+N" tile at the end of an [`AvatarGroup`]: `muted` with `muted-foreground` text, ringed like the
/// avatars and sized to match them (it follows the group's avatars' `data-size`). Holds text or an icon.
#[component]
pub fn AvatarGroupCount(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    children: Element,
) -> Element {
    let base = attributes!(div { class: "dx-avatar-group-count" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/avatar/style.css") }
        div { ..merged, {children} }
    }
}

The Avatar components display a user's profile picture or fallback initials. Use the composable Avatar, AvatarImage, and AvatarFallback primitives when you need full control, or ImageAvatar for the common image-with-fallback case.

Avatar {
    aria_label: "Jane Doe",
    AvatarImage {
        src: "https://example.com/avatar.png",
        alt: "Jane Doe",
    }
    AvatarFallback { "JD" }
}
ImageAvatar {
    src: "https://example.com/avatar.png",
    alt: "Jane Doe",
    on_state_change: |state: AvatarState| { /* image is loading/loaded/failed */ },
    "JD"
}

size: AvatarImageSize follows shadcn's Nova avatar (size-6 / size-8 / size-10) and is emitted as data-size on the root with shadcn's own values (sm, default, lg). The initials in the fallback follow it (text-xs in the small size, text-sm otherwise).

AvatarImageSizeSizeshadcn size
ExtraSmall24pxsm
Small (default)32pxdefault
Medium40pxlg
ImageAvatar { size: AvatarImageSize::Medium, src: "https://example.com/avatar.png", alt: "Jane Doe", "JD" }

Breaking change (Nova migration). The names are the old ones, the scale is shadcn's. Small is unchanged (32px, still the default), so most call sites need nothing. Medium went from 48px to 40px, Large (64px) is gone because shadcn's scale ends at 40px (use Medium, or set width/height on the avatar yourself for a larger one), and ExtraSmall (24px) is new. The dx-avatar-sm|md|lg classes no longer exist: select a size with [data-size="sm|default|lg"], or override --dx-avatar-size on the root.

The fallback is always muted with muted-foreground text, and a 1px border ring is painted over the picture, as in Nova.

The fallback follows Base UI: it is rendered in every state except loaded (while the picture is loading, after it failed, and when there is no picture) and is replaced only by a picture that has loaded, so there is no blank frame between the two. The prerendered page holds both the <img> and the fallback, and hydration starts from the same state, so nothing mismatches.

AvatarFallback { delay: 600 } (ImageAvatar { fallback_delay: 600 }) holds the fallback back for that many milliseconds, so a picture that loads within it never flashes the fallback. While it is held back the avatar shows its loading placeholder, and the prerendered page has no fallback for it (the timer exists only in the browser). An avatar with no AvatarFallback shows a pulsing placeholder while it loads.

Avatar { aria_label: "Jane Doe",
    AvatarImage { src: "https://example.com/avatar.png", alt: "Jane Doe" }
    AvatarFallback { delay: 600, "JD" }
}

While the image is loading and there is no fallback standing in for it, the placeholder pulses (opacity, 1.5 s, forever). The pulse is paused whenever the avatar is off-screen or the tab is hidden (dioxus_primitives::activity::use_motion_when), and an avatar that is not loading subscribes to nothing. Override animation on .dx-avatar[data-state="loading"] to change or remove it. See Motion for the tokens, reduced motion and the per-component table.

AvatarBadge is a status dot or icon on the avatar's bottom-end corner (absolute right-0 bottom-0, mirrored in RTL), ringed in the page background. It is a child of Avatar, in the same place as AvatarFallback. By default it is primary with primary-foreground ink; override background-color for a status colour. Its size follows the avatar: 8px in the small avatar (an icon is hidden there), 10px by default and 12px in the large one, the icon always 8px. The avatar root does not clip (a clip would cut the badge, which sits outside the circle); the picture and the fallback round themselves to the root's shape.

Avatar { aria_label: "Avery Lin, online",
    AvatarImage { src: "https://example.com/avatar.png", alt: "" }
    AvatarFallback { "AL" }
    AvatarBadge { style: "background-color: var(--dx-success);" }
}

The badge is decorative: put the status in the avatar's aria_label.

AvatarGroup stacks avatars with a -8px overlap (-space-x-2) and rings each one in the page background, so the overlap reads as a cut. End it with AvatarGroupCount for a "+N" tile (or an icon); the tile is muted, round, and takes the size of the avatars in the group (24, 32 or 40px). Give the group an aria_label to name it.

AvatarGroup { aria_label: "Team",
    ImageAvatar { src: "https://example.com/a.png", alt: "Avery Lin", "AL" }
    ImageAvatar { src: "https://example.com/b.png", alt: "Casey Park", "CP" }
    AvatarGroupCount { "+3" }
}
