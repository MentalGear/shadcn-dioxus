/// The props of the styled [`Questionnaire`]: the primitive's, plus `animate_height`.
#[derive(Props, Clone, PartialEq)]
pub struct StyledQuestionnaireProps {
    /// The items, in order: the one place each item's behaviour is declared (see [`ItemDef`]).
    pub items: ReadSignal<Vec<ItemDef>>,
    /// The item to start on and to return to on reset (resume).
    #[props(default)]
    pub default_item: ReadSignal<Option<String>>,
    /// The controlled active item.
    #[props(default)]
    pub item: ReadSignal<Option<String>>,
    /// Called with the item to activate after a navigation.
    #[props(default)]
    pub on_item_change: Option<Callback<String>>,
    /// Called after an item's answer changes.
    #[props(default)]
    pub on_answer_change: Option<Callback<AnswerChange>>,
    /// Letter or number keys that select answers.
    #[props(default)]
    pub shortcuts: ReadSignal<Option<ShortcutMode>>,
    /// Use the browser's constraint validation for filled answers.
    #[props(default)]
    pub native_validation: ReadSignal<bool>,
    /// Called on a submit that passed validation.
    #[props(default)]
    pub on_submit: Option<Callback<FormEvent>>,
    /// Called after a reset restored the default answers and the initial item.
    #[props(default)]
    pub on_reset: Option<Callback<()>>,
    /// Glide the height between items with a CSS transition instead of snapping (default `true`).
    /// Skipped under `prefers-reduced-motion: reduce`. `false` keeps the snap.
    #[props(default = true)]
    pub animate_height: bool,
    /// The form's id; generated when absent.
    #[props(default)]
    pub id: ReadSignal<Option<String>>,
    /// Additional attributes for the `form`.
    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,
    /// The progress, the items and the navigation buttons, in any layout.
    pub children: Element,
}

/// Height animation. Items swap with `hidden`, so the form jumps from one item's natural height to the
/// next. `height: auto` cannot transition (`interpolate-size` is Chromium-only), and `grid-template-rows:
/// 0fr/1fr` animates one collapsing child, not a swap. So a ResizeObserver (which runs after layout,
/// before paint: no flash) pins the old height inline, then sets the new one in pixels and lets the CSS
/// `transition: height` in `style.css` run; the pins are cleared when it ends. Works in every engine.
const HEIGHT_JS: &str = r#"
const id = await dioxus.recv();
const el = document.getElementById(id);
if (!el) return;
const reduce = matchMedia('(prefers-reduced-motion: reduce)');
const h = () => el.getBoundingClientRect().height;
let last = h();
let busy = false;
let timer = 0;
cleanup(() => { clearTimeout(timer); el.style.height = ''; el.style.overflow = ''; });
const ro = new ResizeObserver(() => {
  if (busy) return;
  const next = h();
  const from = last;
  last = next;
  if (reduce.matches || Math.abs(next - from) < 1 || !el.isConnected) return;
  const ms = (() => {
    const d = getComputedStyle(el).transitionDuration.split(',')[0].trim();
    return d.endsWith('ms') ? parseFloat(d) : parseFloat(d) * 1000;
  })();
  if (!(ms > 1)) return;
  busy = true;
  el.style.overflow = 'hidden';
  el.style.height = from + 'px';
  void el.offsetHeight;
  el.style.height = next + 'px';
  timer = setTimeout(() => {
    el.style.height = '';
    el.style.overflow = '';
    busy = false;
    last = h();
  }, ms + 60);
});
ro.observe(el);
cleanup(() => ro.disconnect());
"#;

/// Installs the height animation for the enclosing form (a child, so it can read the form's id).
#[component]
fn HeightAnimation() -> Element {
    let q = use_questionnaire();
    use_js_listeners(move || Some(JsListeners::install(HEIGHT_JS).arg(q.form_id())));
    rsx! {}
}

/// A multi-step questionnaire: a native `<form>` that shows one item at a time (Nova's `cn-questionnaire`,
/// `flex flex-col gap-4`). `items` declares every item's behaviour; the parts below render them.
#[component]
pub fn Questionnaire(props: StyledQuestionnaireProps) -> Element {
    let animate = props.animate_height;
    let base = attributes!(div { class: "dx-questionnaire", "data-dx-root": "", "data-slot": "questionnaire" });
    let merged = merge_attributes(vec![
        base,
        attributes!(div { "data-animate-height": animate.then_some("true") }),
        props.attributes,
    ]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/questionnaire/style.css") }
        primitive::Questionnaire {
            items: props.items,
            default_item: props.default_item,
            item: props.item,
            on_item_change: props.on_item_change,
            on_answer_change: props.on_answer_change,
            shortcuts: props.shortcuts,
            native_validation: props.native_validation,
            on_submit: props.on_submit,
            on_reset: props.on_reset,
            id: props.id,
            attributes: merged,
            if animate {
                HeightAnimation {}
            }
            {props.children}
        }
    }
}

/// The props for the styled navigation buttons.
#[derive(Props, Clone, PartialEq)]
pub struct QuestionnaireActionButtonProps {
    /// The [`Button`] variant (Previous and Skip default to outline, Next and Submit to primary).
    #[props(default)]
    pub variant: Option<ButtonVariant>,
    /// The [`Button`] size.
    #[props(default)]
    pub size: ButtonSize,
    /// Disable the button (opt-in, e.g. until the item is answered).
    #[props(default)]
    pub disabled: bool,
    /// Additional attributes for the `button`.
    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,
    /// Replaces the default text.
    pub children: Element,
}

/// A navigation button as a [`Button`], inside a wrapper that is `hidden` while the button does not apply
/// (the Button's own `display` would otherwise override the button's `hidden`).
fn action(
    action: QuestionnaireAction,
    slot: &'static str,
    props: QuestionnaireActionButtonProps,
) -> Element {
    let state = use_questionnaire_action(action, props.disabled);
    let variant = props.variant.unwrap_or(match action {
        QuestionnaireAction::Previous | QuestionnaireAction::Skip => ButtonVariant::Outline,
        QuestionnaireAction::Next | QuestionnaireAction::Submit => ButtonVariant::Primary,
    });
    let attributes = merge_attributes(vec![
        state.attributes,
        attributes!(button { "data-slot": slot }),
        props.attributes,
    ]);
    let default_label = props.children.as_ref().ok() == VNode::empty().as_ref().ok();
    rsx! {
        span {
            class: "dx-questionnaire-action",
            "data-action": slot,
            hidden: !state.visible,
            Button { variant, size: props.size, attributes,
                if default_label {
                    {action.default_label()}
                } else {
                    {props.children}
                }
            }
        }
    }
}

A multi-step question flow for agent clarification prompts, onboarding and surveys (shadcn's questionnaire, 2026-08). One item shows at a time: single choice on native radios, multiple choice on native checkboxes, a freeform "other" answer, an explicit skip for optional items, previous/next/submit, required and external validation, controlled navigation, resume from saved answers, conditional items, optional letter or number answer shortcuts, and native <form> serialization.

let items = vec![
    ItemDef::new("direction").required().choices(["delegation", "questions"]).freeform(),
    ItemDef::new("detail").choices(["focused", "complete"]),
];

Questionnaire { items, on_submit: move |event: FormEvent| save(event.values()),
    QuestionnaireProgress {}
    QuestionnaireItem { name: "direction",
        QuestionnaireTitle { "What should we prototype next?" }
        QuestionnaireDescription { "Choose a direction or write your own." }
        QuestionnaireChoices {
            QuestionnaireChoice { value: "delegation",
                span { "Delegation" }
                QuestionnaireChoiceDescription { "Show how work moves to a specialist." }
            }
            QuestionnaireChoice { value: "questions", "Question prompts" }
            QuestionnaireInput { aria_label: "Another answer", placeholder: "Type another answer…" }
        }
        QuestionnaireError {}
    }
    QuestionnaireItem { name: "detail", /* ... */ }
    QuestionnaireActions {
        QuestionnairePrevious {}
        QuestionnaireSkip {}
        QuestionnaireNext {}
        QuestionnaireSubmit {}
    }
}

One list declares every item. items: Vec<ItemDef> is the only place an item's behaviour lives: its name, required(), multiple(), disabled(..) (a conditional item), invalid(..) (an external error), its choices in render order (choice, choices, disabled_choice), its freeform answer (freeform(), freeform_kind(InputKind::Email)) and its saved answers (default_choice, default_input). QuestionnaireItem takes only a name and QuestionnaireChoice only a value, so the parts carry no behaviour of their own that could disagree with the list. They can still name something the list lacks: a QuestionnaireItem whose name has no ItemDef, a QuestionnaireChoice whose value is not one of the item's choices, or a QuestionnaireInput on an item with no freeform answer. That is not prevented, it is only reported: a tracing::warn! is logged, and the part renders disabled (an unknown item or choice cannot be selected). Duplicate item names, duplicate choice values, a default choice that is not a choice and an unusable default_item are warned about the same way. Render the choices in the list's order, because shortcuts follow it.

Questionnaire propDefault
itemsrequiredThe ItemDefs, in order. Recompute it from earlier answers for conditional items.
default_itemfirst itemThe item to open on and to return to on reset (resume).
item / on_item_changeuncontrolledControlled active item: navigation calls on_item_change and waits for item.
on_answer_changenoneAnswerChange { item, values, status } after an answer changes.
shortcutsnoneShortcutMode::Letters (A-Z) or ShortcutMode::Numbers (1-9).
animate_heighttrueGlide the height between items with a CSS transition instead of snapping; off under prefers-reduced-motion. false keeps the snap. The Animate height demo flips it live.
native_validationfalseValidate filled answers with the browser (an invalid email input blocks Next).
on_submitnoneA submit that passed validation, with the FormEvent (values(), get(name)).
on_resetnoneAfter a type="reset" button restored the saved answers and item.

The navigation buttons take variant, size (the Button props), disabled and children (their label). use_questionnaire() reads the progress, the active item and each item's status, for a custom progress indicator or navigation.

  • Navigation. Next validates the active item, then moves on and focuses the new item's fieldset. Previous needs no validation. Skip (shown only on an optional item) clears the item and moves on, or submits on the last item. Submit validates every item and returns to the first invalid one, focusing its answer.
  • Validation. A required item needs an answer; an optional item needs an answer or an explicit skip. An item's error shows once validation was attempted and stays live: answering clears it, unchecking the last answer brings it back. invalid(true) is an external error (for example from a schema run in on_submit) that blocks the item until you clear it.
  • Freeform answer. Typing selects the input; in a single-choice item it unchecks the radios. Choosing a radio keeps the text as a draft that is neither submitted nor validated.
  • Conditional items. A disabled item leaves the progress count, the navigation and validation, and submits nothing.
  • Height glide. animate_height pins the form's old and new height inline and lets a CSS height transition run between them, so items of different heights do not make the page jump. The Animate height demo has a switch to compare it with the snap; data-animate-height on the form shows the current state.
  • Resume and reset. default_item and each item's defaults restore a saved draft. A reset button returns to them.

Key
ArrowDown / ArrowUpMove between an item's answers (wrapping). Radios also follow the browser's own radio group. An empty text input lets the arrows leave it.
ArrowRight / ArrowLeftNext / previous item, from the item or a button (not from a radio or a text input). ArrowRight never skips an unanswered item.
EnterOn the selected answer: confirm (next, or submit on the last item). Never submits from an unselected answer.
Command/Control + EnterConfirm from anywhere in the form.
A-Z or 1-9With shortcuts: select (or toggle) the answer with that key.

Under RTL (a DirectionProvider, or the page's RTL toggle) ArrowLeft and ArrowRight swap, in aria-keyshortcuts too. Composition (IME) keystrokes and held-key repeats are ignored. The active item lists its keys in aria-keyshortcuts, each answer lists its own.

Every answer is a real form control, so the browser's FormData is the answer set: a checked radio or checkbox submits its value under the item's name, the selected freeform input submits its text, a skipped or disabled item submits nothing. on_submit receives that event. The form has method="dialog" and its submit is always preventDefaulted, so it never navigates (also before hydration).

QuestionnaireItem renders a fieldset and QuestionnaireTitle its legend. The description and, while the item is invalid, the error (role="alert") are joined to the fieldset's aria-describedby; the item and its answers carry aria-invalid. Choices are native radios and checkboxes inside a label. Progress is a progressbar named "Questionnaire progress" with "Question 2 of 3" as its value text. Inactive items and buttons that do not apply are hidden and inert. A successful navigation focuses the new item; a failed validation focuses its first answer (or the selected freeform input).

Give QuestionnaireInput an accessible name with aria_label, aria_labelledby or a visible label: a placeholder is not a label.
