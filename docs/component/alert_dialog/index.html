use dioxus::prelude::*;
use dioxus_primitives::alert_dialog::{
    self, AlertDialogActionsProps, AlertDialogCancelProps, AlertDialogDescriptionProps,
    AlertDialogTitleProps,
};
use dioxus_primitives::dioxus_attributes::attributes;
use dioxus_primitives::keyboard_inset::{use_keyboard_panel, KeyboardInset};
use dioxus_primitives::merge_attributes;

/// The panel's size, shadcn's `size="default" | "sm"`. `Sm` is a compact, centred panel
/// (`max-width: 20rem` at every viewport width) whose actions share one two-column row.
#[derive(Copy, Clone, PartialEq, Eq, Default, Debug)]
#[non_exhaustive]
pub enum AlertDialogSize {
    #[default]
    Default,
    Sm,
}

impl AlertDialogSize {
    /// The `data-size` value the stylesheet keys on.
    pub fn class(&self) -> &'static str {
        match self {
            AlertDialogSize::Default => "default",
            AlertDialogSize::Sm => "sm",
        }
    }
}

/// The look of an [`AlertDialogAction`]: the confirm button of a destructive prompt (the
/// default fill) or the primary button of an ordinary one.
#[derive(Copy, Clone, PartialEq, Eq, Default, Debug)]
#[non_exhaustive]
pub enum AlertDialogActionVariant {
    #[default]
    Destructive,
    Primary,
}

impl AlertDialogActionVariant {
    /// The `data-style` value the stylesheet keys on (the Button vocabulary).
    pub fn class(&self) -> &'static str {
        match self {
            AlertDialogActionVariant::Destructive => "destructive",
            AlertDialogActionVariant::Primary => "primary",
        }
    }
}

/// The props for the [`AlertDialogAction`] component: the primitive's `on_click` plus a typed `variant`.
#[derive(Props, Clone, PartialEq)]
pub struct AlertDialogActionProps {
    /// The click event handler for the action button.
    #[props(default)]
    pub on_click: Option<EventHandler<MouseEvent>>,
    /// Whether the action is destructive (the default) or an ordinary primary action.
    #[props(default)]
    pub variant: AlertDialogActionVariant,
    /// Additional attributes to extend the action button element.
    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,
    /// The children of the action button.
    pub children: Element,
}

/// The props for the [`AlertDialog`] component: [`AlertDialogRootProps`](alert_dialog::AlertDialogRootProps) plus the panel's
/// `initial_focus` / `final_focus` (an alert dialog's panel is rendered by this wrapper).
#[derive(Props, Clone, PartialEq)]
pub struct AlertDialogProps {
    /// The id of the alert dialog root element.
    pub id: ReadSignal<Option<String>>,
    /// Whether the alert dialog is open by default (uncontrolled).
    #[props(default)]
    pub default_open: bool,
    /// The controlled open state.
    #[props(default)]
    pub open: ReadSignal<Option<bool>>,
    /// Whether the scrim is drawn; see [`AlertDialogRootProps::overlay`](alert_dialog::AlertDialogRootProps::overlay).
    #[props(default = ReadSignal::new(Signal::new(true)))]
    pub overlay: ReadSignal<bool>,
    /// Called when the open state changes.
    #[props(default)]
    pub on_open_change: Callback<bool>,
    /// Where focus lands on open, as a function of the open modality
    /// ([`dioxus_primitives::modality::OpenModality`]); see [`dioxus_primitives::alert_dialog::AlertDialogContentProps::initial_focus`].
    #[props(default)]
    pub initial_focus: Option<dioxus_primitives::modality::FocusTargetFn>,

    /// Where focus goes on close; see [`dioxus_primitives::alert_dialog::AlertDialogContentProps::final_focus`].
    #[props(default)]
    pub final_focus: Option<dioxus_primitives::modality::FocusTargetFn>,

    /// The panel's size (`data-size` on the `<dialog>`). Defaults to [`AlertDialogSize::Default`].
    #[props(default)]
    pub size: AlertDialogSize,

    /// Additional attributes for the root element.
    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,
    /// The children of the alert dialog.
    pub children: Element,
}

#[component]
pub fn AlertDialog(props: AlertDialogProps) -> Element {
    let base = attributes!(div {
        class: "dx-alert-dialog-backdrop",
        "data-dx-scrim": "",
    });
    let merged = merge_attributes(vec![base, props.attributes]);
    // The panel follows the on-screen keyboard (`primitives/src/keyboard_inset.rs`; `style.css`
    // consumes `--dx-keyboard-inset`).
    let panel = use_keyboard_panel(None);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/alert_dialog/style.css") }
        alert_dialog::AlertDialogRoot {
            id: props.id,
            default_open: props.default_open,
            open: props.open,
            overlay: props.overlay,
            on_open_change: props.on_open_change,
            attributes: merged,
            alert_dialog::AlertDialogContent {
                id: panel.id(),
                class: "dx-alert-dialog".to_string(),
                attributes: attributes!(div { "data-size": props.size.class() }),
                initial_focus: props.initial_focus,
                final_focus: props.final_focus,
                {props.children}
                KeyboardInset { panel }
            }
        }
    }
}

/// An icon or image above the title (shadcn's `AlertDialogMedia`): a rounded muted tile that
/// centres whatever it holds, sized by the dialog's `size`.
#[component]
pub fn AlertDialogMedia(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    /// The icon or image shown in the tile.
    children: Element,
) -> Element {
    let base = attributes!(div { class: "dx-alert-dialog-media", "data-slot": "alert-dialog-media" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/alert_dialog/style.css") }
        div { ..merged, {children} }
    }
}

The AlertDialog primitive provides an accessible, composable modal dialog for critical user confirmations (such as destructive actions).

let mut open = use_signal(|| false);
rsx! {
    button { onclick: move |_| open.set(true), type: "button", "Show Alert Dialog" }
    AlertDialog { open: open(), on_open_change: move |v| open.set(v),
        AlertDialogTitle { "Title" }
        AlertDialogDescription { "Description" }
        AlertDialogActions {
            AlertDialogCancel { "Cancel" }
            AlertDialogAction { on_click: move |_| { /* destructive action */ }, "Confirm" }
        }
    }
}

Components

  • AlertDialog: Provides context, manages open state, renders the backdrop and the content panel.
  • AlertDialogTitle: The dialog's heading.
  • AlertDialogDescription: Additional description for the dialog.
  • AlertDialogActions: Container for action buttons.
  • AlertDialogAction: Main action button (e.g., confirm/delete). Closes dialog and calls optional on_click.
  • AlertDialogCancel: Cancel/close button. Closes dialog and calls optional on_click.

A modal alert dialog dims the page behind it with the shared overlay scrim: the dialog's own ::backdrop, the same 10% black and 4px blur on every modal overlay. Pass overlay: false to keep the modality (focus trap, inert page; an alert dialog never dismisses on a click outside, only on Escape or its own buttons) without the dim and the blur; the <dialog> then carries data-dx-overlay="off".

Closing keeps the alert dialog a modal -- in the top layer, at the same size and place -- until its exit animation has played, and only then closes it. That holds in every browser; no stylesheet needs an engine-specific keep-alive.

An alert dialog is not closed the instant open turns false. It stays open until its own exit animation and its backdrop's have settled (at most 1500 ms) and only then calls close(), so the exit is always played in full. The scrim and the panel share --dx-overlay-duration and --dx-overlay-ease (defaults --dx-motion-duration-slow and --dx-motion-ease). Under prefers-reduced-motion: reduce both durations become --dx-motion-duration-reduced. Nothing loops. See Motion for the tokens, reduced motion and the per-component table.

initial_focus and final_focus take a function of the open modality (OpenModality::{Keyboard, Mouse, Touch, Pen}, Base UI's initialFocus(openType)) that returns a FocusTarget: Default, Panel, Selector("#id") or, for final_focus, None. They are inputs to the one open-focus policy (dxOpenFocus / dxCloseFocus, primitives/src/modality.rs), not a second code path: the function is called once per modality at render, the answers ride on the panel as data-dx-initial-focus / data-dx-final-focus, and the policy reads the entry for the modality of the open. A selector that matches nothing falls back to the built-in policy (keyboard and mouse: the first control; touch and pen: the panel, so no software keyboard rises). An explicit target is honoured as given, so return Panel for Touch yourself when it names a field.

AlertDialog {
    initial_focus: Callback::new(|m: OpenModality| match m {
        OpenModality::Touch => FocusTarget::Panel,
        _ => FocusTarget::selector("#name"),
    }),
    final_focus: Callback::new(|_| FocusTarget::Default),
}

final_focus defaults to the element that held focus when the panel opened: the trigger for a click open, the previously focused element for a programmatic open. The opener is remembered in exactly one place (dxOpenFocus), for every overlay kind. See the focus_target variant.

size: AlertDialogSize::Sm (shadcn's size="sm") renders a compact, centred panel, max-width: 20rem at every viewport width, with the two actions on one two-column row; the panel carries data-size. AlertDialogMedia puts an icon (or image) tile above the title, 64px with a 32px icon, or 40px / 24px in Sm.

AlertDialogAction { variant } takes AlertDialogActionVariant::Destructive (the default, a red confirm button) or Primary for a prompt that is not destructive; the button carries data-style, the Button vocabulary.

AlertDialog { size: AlertDialogSize::Sm, open: open(), on_open_change: move |v| open.set(v),
    AlertDialogMedia { BluetoothConnected {} }
    AlertDialogTitle { "Allow Bluetooth pairing?" }
    AlertDialogDescription { "Your headphones want to pair with this device." }
    AlertDialogActions {
        AlertDialogCancel { "Not now" }
        AlertDialogAction { variant: AlertDialogActionVariant::Primary, "Pair" }
    }
}

Examples

Alternative examples for common configurations.

small

size: AlertDialogSize::Sm gives a compact, centred panel whose two actions share one row.

media

AlertDialogMedia puts an icon tile above the title, shown here in the default size and in Sm.

non destructive

variant: AlertDialogActionVariant::Primary makes the action a normal confirm button, so a dialog that is not destructive is not painted red.

overlay

overlay: false keeps the dialog modal and drops only the dim and blur behind it.

focus target

initial_focus keeps the destructive action from taking the default focus: a keyboard open lands on Cancel, a mouse or touch open on the panel.
