use dioxus::prelude::*;
use dioxus_icons::lucide::{Check, ChevronsUpDown, X};
use dioxus_primitives::combobox::{
    self, default_combobox_filter, ComboboxEmptyProps, ComboboxOptionProps,
};
use dioxus_primitives::{dioxus_attributes::attributes, merge_attributes};

#[derive(Props, Clone, PartialEq)]
pub struct ComboboxProps<T: Clone + PartialEq + 'static = String> {
    /// The controlled selected value. A signal holding `None` means no option is selected.
    #[props(default)]
    pub value: Option<ReadSignal<Option<T>>>,

    /// The initially selected value when uncontrolled.
    #[props(default)]
    pub default_value: Option<T>,

    /// A selected value's display text, shown in the input. Required: it is read at render time, so a
    /// preselected value's text is in the server-rendered HTML (the option registry is filled by a
    /// client-only effect and can never supply it).
    pub item_to_label: Callback<T, String>,

    /// Called with the new value (`None` when the selection is cleared).
    #[props(default)]
    pub on_value_change: Callback<Option<T>>,

    /// Whether the combobox is disabled.
    #[props(default)]
    pub disabled: ReadSignal<bool>,

    /// The controlled open state of the list.
    #[props(default)]
    pub open: ReadSignal<Option<bool>>,

    /// Whether the list starts open when uncontrolled.
    #[props(default)]
    pub default_open: ReadSignal<bool>,

    /// Called when the list opens or closes.
    #[props(default)]
    pub on_open_change: Callback<bool>,

    /// The controlled text in the input.
    #[props(default)]
    pub query: ReadSignal<Option<String>>,

    /// The initial text in the input when uncontrolled.
    #[props(default)]
    pub default_query: ReadSignal<String>,

    /// Called with the new text whenever the user types.
    #[props(default)]
    pub on_query_change: Callback<String>,

    /// Whether arrow-key navigation wraps from the last option to the first.
    #[props(default = ReadSignal::new(Signal::new(true)))]
    pub roving_loop: ReadSignal<bool>,

    /// Custom filter, called with `(query, option_text_value)`; returns whether the option stays visible.
    #[props(default = Callback::new(|(q, t): (String, String)| default_combobox_filter(&q, &t)))]
    pub filter: Callback<(String, String), bool>,

    /// Placeholder text of the input.
    #[props(default)]
    pub placeholder: ReadSignal<String>,

    /// Accessible name for the input.
    #[props(default)]
    pub aria_label: Option<String>,

    /// Accessible name for the listbox of options.
    #[props(default)]
    pub list_aria_label: Option<String>,

    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,

    /// The combobox parts: trigger, list, groups and options.
    pub children: Element,
}

#[component]
pub fn Combobox<T: Clone + PartialEq + 'static>(props: ComboboxProps<T>) -> Element {
    let base = attributes!(div { class: "dx-combobox", "data-dx-root": "" });
    let merged = merge_attributes(vec![base, props.attributes]);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/combobox/style.css") }
        combobox::Combobox {
            value: props.value,
            default_value: props.default_value,
            item_to_label: props.item_to_label,
            on_value_change: props.on_value_change,
            disabled: props.disabled,
            open: props.open,
            default_open: props.default_open,
            on_open_change: props.on_open_change,
            query: props.query,
            default_query: props.default_query,
            on_query_change: props.on_query_change,
            roving_loop: props.roving_loop,
            filter: props.filter,
            attributes: merged,
            div { class: "dx-combobox-input-wrapper",
                combobox::ComboboxInput {
                    class: "dx-combobox-input",
                    placeholder: props.placeholder,
                    aria_label: props.aria_label.clone(),
                }
                ChevronsUpDown {
                    class: "dx-combobox-expand-icon",
                    size: "16px",
                }
            }
            combobox::ComboboxList {
                class: "dx-combobox-list",
                "data-dx-surface": "floating",
                aria_label: props.list_aria_label.clone(),
                {props.children}
            }
        }
    }
}

#[derive(Props, Clone, PartialEq)]

    #[props(default)]
    pub disabled: ReadSignal<bool>,

    pub open: ReadSignal<Option<bool>>,

    /// Whether the list starts open when uncontrolled.
    #[props(default)]
    pub default_open: ReadSignal<bool>,

    /// Called when the list opens or closes.
    #[props(default)]
    pub on_open_change: Callback<bool>,

    /// The controlled text in the input.
    #[props(default)]
    pub query: ReadSignal<Option<String>>,

    /// The initial text in the input when uncontrolled.
    #[props(default)]
    pub default_query: ReadSignal<String>,

    /// Called with the new text whenever the user types.
    #[props(default)]
    pub on_query_change: Callback<String>,

    /// Whether arrow-key navigation wraps from the last option to the first.
    #[props(default = ReadSignal::new(Signal::new(true)))]
    pub roving_loop: ReadSignal<bool>,

    /// Custom filter, called with `(query, option_text_value)`; returns whether the option stays visible.
    #[props(default = Callback::new(|(q, t): (String, String)| default_combobox_filter(&q, &t)))]
    pub filter: Callback<(String, String), bool>,

    /// Placeholder text of the input.
    #[props(default)]
    pub placeholder: ReadSignal<String>,

    /// Accessible name for the input.
    #[props(default)]
    pub aria_label: Option<String>,

    /// Accessible name for the listbox of options.
    #[props(default)]
    pub list_aria_label: Option<String>,

    /// Shows a button that clears every chip while anything is selected.
    #[props(default)]
    pub show_clear: bool,

    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,

    /// The combobox parts: the options, groups and the empty state.
    pub children: Element,
}

/// A combobox that selects several options and shows them as removable chips.
///
/// Backspace in the empty input removes the last chip; the list stays open while options are toggled.
#[component]
pub fn ComboboxMulti<T: Clone + PartialEq + ToString + 'static>(
    props: ComboboxMultiProps<T>,
) -> Element {
    let base = attributes!(div { class: "dx-combobox dx-combobox-multi", "data-dx-root": "" });
    let merged = merge_attributes(vec![base, props.attributes]);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/combobox/style.css") }
        combobox::ComboboxMulti {
            values: props.values,
            default_values: props.default_values,
            item_to_label: props.item_to_label,
            on_values_change: props.on_values_change,
            name: props.name,
            disabled: props.disabled,
            open: props.open,
            default_open: props.default_open,
            on_open_change: props.on_open_change,
            query: props.query,
            default_query: props.default_query,
            on_query_change: props.on_query_change,
            roving_loop: props.roving_loop,
            filter: props.filter,
            attributes: merged,
            combobox::ComboboxChips {
                class: "dx-combobox-chips",
                chip_class: "dx-combobox-chip",
                remove_class: "dx-combobox-chip-remove",
                remove_icon: rsx! {
                    X { size: "12px" }
                },
                combobox::ComboboxInput {
                    class: "dx-combobox-chips-input",
                    placeholder: props.placeholder,
                    aria_label: props.aria_label.clone(),
                }
                if props.show_clear {
                    combobox::ComboboxClear {
                        class: "dx-combobox-clear",
                        X { size: "14px" }
                    }
                }
            }
            combobox::ComboboxList {
                class: "dx-combobox-list",
                "data-dx-surface": "floating",
                aria_label: props.list_aria_label.clone(),
                {props.children}
            }
        }
    }
}

The Combobox component is an autocomplete input with a filterable popup list.

Filtering preserves the order defined by the rendered ComboboxOption elements and their index props. If you want query-dependent ranking, control query, sort your item data in user code, render the options in that sorted order, and assign indexes from the sorted list.

Selecting several options (chips, clear button, form value) is ComboboxMulti: its demos come first under Examples below, and its API is in "Multiple selection".

The input shows the selected option's label. Clicking, focusing or arrowing into it never rewrites that text: the caret lands where the user clicked, the popup lists every option, and the selected one is marked and scrolled into view. Filtering starts with the first edit, and from then on the input's own text is the query. Escape, blurring, or picking an option ends the edit and puts the selected label back.

let mut value = use_signal(|| None::<String>);
let mut query = use_signal(String::new);

Combobox::<String> {
    value: Some(value.into()),
    item_to_label: |value: String| label_of(&value), // required: the input's text for the value, read at render time
    on_value_change: move |next: Option<String>| {
        value.set(next);
    },
    query: Some(query()),
    on_query_change: move |next| query.set(next),
    placeholder: "Select framework...",
    aria_label: "Select framework",
    list_aria_label: "Frameworks",
    ComboboxEmpty { "No framework found." }
    ComboboxOption::<String> {
        index: 0usize,
        value: "next".to_string(),
        text_value: "Next.js",
        "Next.js"
    }
}

item_to_label (required on Combobox and ComboboxMulti, like Base UI's itemToStringLabel) is the selected value's display text. The input's value reads it at render time, never the option registry (which fills in a client-only effect), so a preselected Combobox carries value="Next.js" in the server-rendered HTML instead of an empty input until hydration, and the input cannot flash empty while the list re-registers its options.

ComboboxMulti selects several options and shows them as removable chips (shadcn's multiple + chips combobox). It takes and reports a Vec<T> (values / default_values / on_values_change) where Combobox takes Option<T>: the two roots have different types, so the operations that only exist on a set (remove one, clear all, Backspace on an empty input, chips) cannot be reached from a single-select root, and a single root's callback can never see a list. Both share the same input, list, options and empty state.

ComboboxMulti::<String> {
    default_values: vec!["next".to_string()],
    item_to_label: |value: String| label_of(&value), // required: the chip's text, read at render time
    name: Some("frameworks".to_string()), // one form entry per selected value
    show_clear: true,                     // a clear button while anything is selected
    placeholder: "Choose frameworks...",
    aria_label: "Frameworks",
    ComboboxEmpty { "No framework found." }
    ComboboxOption::<String> { index: 0usize, value: "next".to_string(), text_value: "Next.js", "Next.js" }
}
  • item_to_label (required, like Base UI's itemToStringLabel) is a selected value's display text. Chips and their remove buttons' accessible names read it, never the option registry, so a preselected chip carries its text in the server-rendered HTML instead of rendering empty until hydration.
  • The list stays open while options are toggled; picking a selected option removes it. The typed text is the query only and is cleared after each pick and when the popup closes.
  • The listbox carries aria-multiselectable="true" (the single list never does); every option keeps aria-selected.
  • Keyboard: Backspace in the empty input removes the last chip. The "previous" arrow (ArrowLeft in LTR, ArrowRight in RTL) moves from the empty input onto the chips' remove buttons, the arrows move between them (the "next" arrow past the last chip returns to the input), Enter/Space/Backspace/Delete remove the focused chip and return focus to the input. The chips field is the popup's anchor and width.
  • Forms: with name, every selected value is submitted under it (FormData.getAll(name)), as the value's ToString; <form reset> restores default_values. Values need T: ToString.
  • Set --dx-combobox-width on the root to size the field and the list together.

The list fades and scales in and out over --dx-floating-duration (the shared dx-pop-in/dx-pop-out). Retune with those tokens. Reduced motion swaps in --dx-motion-duration-reduced. Nothing loops. See Motion for the tokens, reduced motion and the per-component table.

Web arm (Migration A slice 3/3, final): promote the listbox to the top layer via popover="manual", anchored to ComboboxInput below/ start-aligned -- the same top-layer fix already shipped for Tooltip/HoverCard/Popover/DropdownMenu/Menubar/Select (docs/plan.md Phase 4.4, Migration A slices 1-3), matching this component's pre-migration top: calc(100% + 0.25rem); left: 0 CSS (../../../../preview/src/components/combobox/style.css).

manual, reversed from auto by execution

This slice's first cut used auto -- Combobox has no Select-style DOM-focus/blur delicacy to worry about (ComboboxInput keeps focus throughout via an aria-activedescendant model, never blurring into this listbox), so auto's light dismiss looked like free, strictly additive insurance on top of ComboboxInput's own onblur/onkeydown dismissal. Confirmed wrong by execution: combobox.spec.ts's "dynamic option removal updates filtering and keyboard selection" test went red -- clicking an external "Toggle SvelteKit" button while the list was open (a supported pattern: that button's own onpointerdown already calls prevent_default() specifically so it does not blur/close the combobox) unexpectedly closed the list anyway. WHATWG light dismiss classifies any pointerdown outside the popover's own DOM subtree as "outside," full stop -- confirmed by execution that neither prevent_default() on that pointerdown nor a popovertarget association from ComboboxInput to this listbox stops it (auto has no notion of "this button is allowed to interact with me without closing me" beyond "is it a descendant"). That is fatal specifically for Combobox's own supported use case -- an external control manipulating state while the list stays open -- in a way it is not for Select (whose own suite has no such case, and for which "any outside click closes it" is the same behavior a plain HTML <select> already has). manual sidesteps the whole question: WHATWG HTML never light-dismisses a manual popover, so ComboboxInput's own onblur/onkeydown stay the only dismissal path -- unchanged from pre-migration, and immune to this class of external-click interference -- exactly ContextMenuContentRendered's identical manual reasoning (context_menu.rs).

The one behavior this reversal gives up: auto's light dismiss would additionally have closed the list on a click over a non-focusable area (one ComboboxInput's onblur alone cannot see, since nothing there ever takes focus). No combobox.spec.ts case depends on that gap being closed, so trading it away for the external-control case above is a clear net win, not a compromise made blind.

Animation: still does not use crate::top_layer::use_popover_sync bound to the raw open signal -- see crate::top_layer::use_popover_shown_while_mounted's doc for the animate-out race that binding would otherwise cause with crate::use_animated_open (ComboboxList, above). Confirmed by execution to matter here specifically: combobox.spec.ts's "keeps filtered options during keyboard close animation" test (an Enter-key, script-driven close) requires the data-state="closed" element to stay visible through its exit animation, which binding showPopover()/hidePopover() to the raw open signal breaks (see the linked doc for the exact mechanism) -- this reasoning is orthogonal to the auto-vs-manual reversal above and holds either way.
