use dioxus::prelude::*;
use dioxus_primitives::dioxus_attributes::attributes;
use dioxus_primitives::merge_attributes;

/// The props every glyph takes.
#[derive(Clone, PartialEq, Props)]
pub struct IconProps {
    /// The glyph's width and height: a number of pixels (`"16"`) or any CSS length (`"1rem"`).
    /// Defaults to 24.
    #[props(into, default = "24".to_string())]
    pub size: String,
    #[props(extends = SvgAttributes)]
    pub attributes: Vec<Attribute>,
}

/// The `<svg>` every glyph shares: the Lucide defaults first, so an attribute the caller passes
/// (`stroke`, `class`, `aria-hidden`, ..) comes after and wins.
#[component]
fn Svg(
    /// The glyph's width and height.
    size: String,
    /// The caller's attributes.
    attributes: Vec<Attribute>,
    /// The glyph's shapes.
    children: Element,
) -> Element {
    let stroke = if attributes.iter().any(|a| a.name == "stroke") { None } else { Some("currentColor") };
    let defaults = attributes!(svg {
        xmlns: "http://www.w3.org/2000/svg",
        view_box: "0 0 24 24",
        fill: "none",
        stroke_width: "2",
        stroke_linecap: "round",
        stroke_linejoin: "round",
    });
    let merged = merge_attributes(vec![defaults, attributes]);
    rsx! {
        svg {
            width: "{size}",
            height: "{size}",
            stroke,
            ..merged,
            {children}
        }
    }
}

macro_rules! glyph {
    ($(#[$doc:meta])* $name:ident { $($shape:tt)* }) => {
        $(#[$doc])*
        #[component]
        #[allow(non_snake_case, reason = "a glyph component is named like the icon (PascalCase), as Dioxus components are")]
        pub fn $name(props: IconProps) -> Element {
            rsx! {
                Svg { size: props.size, attributes: props.attributes, $($shape)* }
            }
        }
    };
}

Every glyph renders one inline <svg> on a 24 x 24 view box with a 2px round stroke in currentColor, so it follows the text colour of its parent. size sets the width and height (a pixel count or any CSS length, 24 by default); every other SVG attribute you pass (class, stroke, aria-hidden, data-*) lands on the <svg> and wins over the default.

IconProps Styled component

The props every glyph takes.

PropTypeDefaultDescription
sizeString"24".to_string()

The glyph's width and height: a number of pixels ("16") or any CSS length ("1rem"). Defaults to 24.

Plus the <SvgAttributes> element's attributes.

Svg Styled component

The <svg> every glyph shares: the Lucide defaults first, so an attribute the caller passes (stroke, class, aria-hidden, ..) comes after and wins.

PropTypeDefaultDescription
sizeStringrequired

The glyph's width and height.

attributesVec<Attribute>required

The caller's attributes.

childrenElementempty

The glyph's shapes.

Component-Specific

What is particular to this component.

Add a glyph! block to component.rs with the shapes copied from the Lucide icon of the same name (https://lucide.dev). Lucide is ISC-licensed; the notice at the top of the file stays with it, plus the MIT notice of the Feather project for the glyphs Lucide derived from it.

The file holds only the glyphs the components use. An app that wants any other Lucide icon adds the dioxus-icons crate itself and imports from dioxus_icons::lucide; nothing here depends on it.

A glyph carries no name. Inside a button that has a text name, pass "aria-hidden": "true"; for an icon-only control, put the accessible name on the control (aria_label), not on the glyph.
