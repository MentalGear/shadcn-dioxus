fn fixed_date(year: i32, month: Month, day: u8) -> Date {
    Date::from_calendar_date(year, month, day).expect("valid fixed date")
}

fn weekday_abbreviation(weekday: Weekday) -> &'static str {
    match weekday {
        Weekday::Monday => "Mo",
        Weekday::Tuesday => "Tu",
        Weekday::Wednesday => "We",
        Weekday::Thursday => "Th",
        Weekday::Friday => "Fr",
        Weekday::Saturday => "Sa",
        Weekday::Sunday => "Su",
    }
}

/// How a month's caption is drawn (react-day-picker's `captionLayout`, shadcn's `Calendar`).
///
/// `Label` (the default) is the plain text "October 2026" between the previous/next arrows. The
/// three dropdown layouts swap the month and/or the year for a native `<select>` (labelled "Month"
/// and "Year"), so a far-away date is one pick instead of many clicks on the arrows; the part that
/// is not a dropdown stays plain text beside it.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
pub enum CaptionLayout {
    /// Plain "Month Year" text.
    #[default]
    Label,
    /// A month select and a year select.
    Dropdown,
    /// A month select; the year is plain text.
    DropdownMonths,
    /// A year select; the month is plain text.
    DropdownYears,
}

impl CaptionLayout {
    fn has_month_dropdown(self) -> bool {
        matches!(self, Self::Dropdown | Self::DropdownMonths)
    }

    fn has_year_dropdown(self) -> bool {
        matches!(self, Self::Dropdown | Self::DropdownYears)
    }
}

#[derive(Props, Clone, PartialEq)]
pub struct CalendarProps {
    /// The selected date
    #[props(default)]
    pub selected_date: ReadSignal<Option<Date>>,

    /// Callback when selected date changes
    #[props(default)]
    pub on_date_change: Callback<Option<Date>>,

    /// Callback when display weekday
    #[props(default = Callback::new(|weekday: Weekday| weekday_abbreviation(weekday).to_string()))]
    pub on_format_weekday: Callback<Weekday, String>,

    /// Callback when display month
    #[props(default = Callback::new(|month: Month| month.to_string()))]
    pub on_format_month: Callback<Month, String>,

    /// The month being viewed
    #[props(default = ReadSignal::new(Signal::new(UtcDateTime::now().date())))]
    pub view_date: ReadSignal<Date>,

    /// The current date (used for highlighting today)
    #[props(default = UtcDateTime::now().date())]
    pub today: Date,

    /// Callback when view date changes
    #[props(default)]
    pub on_view_change: Callback<Date>,

    /// Whether the calendar is disabled
    #[props(default)]
    pub disabled: ReadSignal<bool>,

    /// First day of the week
    #[props(default = Weekday::Sunday)]
    pub first_day_of_week: Weekday,

    /// How the month caption is drawn: plain text (default) or month/year dropdowns
    #[props(default)]
    pub caption_layout: CaptionLayout,

    /// Show a week-number column (ISO weeks with a Monday `first_day_of_week`, otherwise the week
    /// holding January 1st is week 1)
    #[props(default)]
    pub show_week_number: bool,

    /// Lower limit of the range of available dates
    #[props(default = fixed_date(1925, Month::January, 1))]
    pub min_date: Date,

    /// Upper limit of the range of available dates
    #[props(default = fixed_date(2050, Month::December, 31))]
    pub max_date: Date,

    /// Specify how many months are visible at once
    #[props(default = 1)]
    pub month_count: u8,

    /// Unavailable dates
    #[props(default)]
    pub disabled_ranges: ReadSignal<Vec<calendar::DateRange>>,

    /// Additional attributes to extend the calendar element
    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,
}

#[component]
pub fn Calendar(props: CalendarProps) -> Element {
    let month_count = props.month_count.max(1);

    rsx! {
        CalendarRoot {
            selected_date: props.selected_date,
            on_date_change: props.on_date_change,
            on_format_weekday: props.on_format_weekday,
            on_format_month: props.on_format_month,
            view_date: props.view_date,
            today: props.today,
            on_view_change: props.on_view_change,
            disabled: props.disabled,
            first_day_of_week: props.first_day_of_week,
            show_week_number: props.show_week_number,
            min_date: props.min_date,
            max_date: props.max_date,
            disabled_ranges: props.disabled_ranges,
            attributes: props.attributes,
            for offset in 0..month_count {
                CalendarMonthView {
                    key: "{offset}",
                    offset,
                    month_count,
                    caption_layout: props.caption_layout,
                }
            }
        }
    }
}

// docs/backlog.md row 51: the stylesheet `Link` lives HERE, on the
// `pub(crate)` root each of this module's real entry points funnels
// through, not on `Calendar`/`RangeCalendar` above -- `date_picker`
// (`../date_picker/component.rs`) composes `CalendarRoot`/
// `RangeCalendarRoot` directly (`super::super::calendar::*`), bypassing
// `Calendar`/`RangeCalendar` entirely, since `DatePickerCalendar`'s
// `calendar: fn(T) -> Element` prop needs a bare component function to call,
// not a wrapper that also takes a `month_count` prop. Before this fix the
// Link sat only on `Calendar`/`RangeCalendar`, so that composition path
// never rendered it and the calendar's CSS silently never reached
// `/component/?name=date_picker&` -- root-caused by comparing `Calendar`'s
// own route (styled) against `date_picker`'s composition of it (unstyled).
// Putting it here instead -- matching `popover/component.rs`'s pattern of a
// `Link` on every one of ITS composable entry points, not just its "main"
// one -- means every path that ends up rendering the primitive (this
// module's own `Calendar`/`RangeCalendar`, and any other themed component
// that composes `CalendarRoot`/`RangeCalendarRoot` the way `date_picker`
// does) gets the stylesheet for free, by construction; `document::Link`'s
// `(href, rel)` dedup makes the now-redundant render from `Calendar`'s own
// `document::Link` (removed above) unnecessary rather than merely harmless.
#[component]
pub(crate) fn CalendarRoot(props: calendar::CalendarProps) -> Element {
    let base = attributes!(div {
        class: "dx-calendar"
    });
    let merged = merge_attributes(vec![base, props.attributes]);

    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/calendar/style.css") }
        calendar::Calendar {
            selected_date: props.selected_date,
            on_date_change: props.on_date_change,
            on_format_weekday: props.on_format_weekday,
            on_format_month: props.on_format_month,
            view_date: props.view_date,
            today: props.today,
            on_view_change: props.on_view_change,
            disabled: props.disabled,
            first_day_of_week: props.first_day_of_week,
            show_week_number: props.show_week_number,
            min_date: props.min_date,
            max_date: props.max_date,
            disabled_ranges: props.disabled_ranges,
            attributes: merged,
            {props.children}
        }
    }
}

#[component]
pub(crate) fn CalendarMonthView(
    offset: u8,
    month_count: u8,
    #[props(default)] caption_layout: CaptionLayout,
) -> Element {
    let show_previous = offset == 0;
    let show_next = offset.saturating_add(1) == month_count;

    rsx! {
        CalendarView { offset,
            CalendarHeader {
                CalendarNavigation {
                    if show_previous {
                        CalendarPreviousMonthButton {}
                    }
                    CalendarCaption { layout: caption_layout }
                    if show_next {
                        CalendarNextMonthButton {}
                    }
                }
            }
            CalendarGrid {}
        }
    }
}

// react-day-picker's `MonthCaption`: `label` is one live-region text node; the dropdown layouts put
// a native select (or, for the part that is not a dropdown, plain text) per field and keep the full
// "October 2026" as a visually hidden live region, because a select's own name ("Month") does not
// say which month a prev/next click landed on.
#[component]
fn CalendarCaption(layout: CaptionLayout) -> Element {
    if layout == CaptionLayout::Label {
        let attributes = attributes!(div {
            class: "dx-calendar-month-title",
            role: "status",
            aria_live: "polite",
        });
        return rsx! {
            calendar::CalendarMonthTitle { attributes }
        };
    }

    let text = attributes!(span { class: "dx-calendar-caption-text" });
    let hidden = attributes!(div {
        class: "dx-calendar-sr-only",
        role: "status",
        aria_live: "polite",
    });
    rsx! {
        if layout.has_month_dropdown() {
            CalendarSelectMonth {}
        } else {
            calendar::CalendarSelectMonthValue { attributes: text.clone() }
        }
        if layout.has_year_dropdown() {
            CalendarSelectYear {}
        } else {
            calendar::CalendarSelectYearValue { attributes: text }
        }
        calendar::CalendarMonthTitle { attributes: hidden }
    }
}

#[component]
fn CalendarGrid(
    #[props(default)] id: Option<String>,
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
) -> Element {
    let grid = calendar::use_calendar_grid();
    let show_week_number = grid.show_week_number();
    // The root's owned `data-*` hook for the week-number column's wider grid (style.css).
    let state = attributes!(div { "data-week-numbers": show_week_number });
    let attributes = merge_attributes(vec![state, attributes]);

    rsx! {
        CalendarGridRoot { id, attributes,
            CalendarGridHead {
                CalendarGridHeaderRow {
                    if show_week_number {
                        CalendarGridWeekNumberHeader {}
                    }
                    for weekday in grid.weekdays().iter().cloned() {
                        CalendarGridDayHeader {
                            key: "{weekday.weekday():?}",
                            weekday: weekday.weekday(),
                            {weekday.label().to_string()}
                        }
                    }
                }
            }
            CalendarGridBody {
                for (week , week_number) in grid.weeks().iter().zip(grid.week_numbers().iter().copied()) {
                    CalendarGridWeek {
                        if show_week_number {
                            CalendarGridWeekNumber { week_number }
                        }
                        for date in week.iter().copied() {
                            CalendarGridCell {
                                key: "{date}",
                                CalendarDay { date }
                            }
                        }
                    }
                }
            }
        }
    }
}

Usage notes

The Calendar component is used to display a calendar interface, allowing users to select dates. It provides a grid layout of days for a specific month and year.

Component Structure

Calendar {
    // The currently selected date in the calendar (if any).
    selected_date,
    on_date_change: |date: Option<CalendarDate>| {
        // This callback is triggered when a date is selected in the calendar.
        // The date parameter contains the selected date.
    },
    // The current view date of the calendar, which determines the month and year displayed.
    view_date,
    on_view_change: |date: CalendarDate| {
        // This callback is triggered when the view date changes.
        // The date parameter contains the new view date.
    },
    // Optional number of pre-composed month views to render.
    month_count: 1,
    // How the month caption is drawn: Label (default), Dropdown, DropdownMonths, DropdownYears.
    caption_layout: CaptionLayout::Label,
    // Show a week-number column.
    show_week_number: false,
}

The styled Calendar and RangeCalendar render a complete month view by default, including the header, navigation, caption (see caption_layout), and date grid.

Direction / RTL

Calendar/RangeCalendar accept a dir: Option<Direction> prop. Under RTL, the day grid's ArrowLeft/ArrowRight swap (ArrowLeft moves to the next day, ArrowRight to the previous one); ArrowUp/ArrowDown (+/-7 days) are unaffected. No Radix Calendar exists to cite (shadcn's own wraps react-day-picker) -- this follows the same arrow-key convention as every other RTL-aware component here, and matches standard calendar-widget RTL practice generally. See the rtl variant.

Multiple months

month_count renders that many months in one calendar. They sit side by side in a flex row with a gap (the previous-month button on the first month, the next-month button on the last) and wrap onto their own rows when the row is wider than the space available -- on a phone, or in a narrow card, two months stack vertically instead of overflowing. Inside a DatePicker/DateRangePicker popover, the panel grows to fit the months. See the multi_month variant.

Caption layout

caption_layout chooses how each month's caption is drawn (react-day-picker's captionLayout, which shadcn's Calendar passes through):

CaptionLayoutCaption
Label (default)plain text, "October 2026", between the previous and next arrows
Dropdowna month select and a year select
DropdownMonthsa month select; the year is plain text
DropdownYearsa year select; the month is plain text

The selects are native <select> elements named "Month" and "Year", so they keep the platform keyboard and touch behaviour (the options are limited by min_date / max_date). The caption text is a polite live region; in the dropdown layouts the full "October 2026" is kept as a visually hidden live region too, so a screen reader hears where an arrow click landed. The label (and the dropdown option text) goes through on_format_month. See the caption variant for all four side by side.

The default changed from "always both dropdowns" to Label, following shadcn: pass caption_layout: CaptionLayout::Dropdown to keep the old look. DatePicker and DateRangePicker use the default.

Week numbers

show_week_number: true adds a column of week numbers in front of every week row, and an empty header cell above it (react-day-picker's showWeekNumber). With first_day_of_week: Weekday::Monday the numbers are ISO 8601 weeks (week 1 holds the year's first Thursday); with any other first day, week 1 is the week that holds January 1st, which is what react-day-picker prints for en-US. Each number is a rowheader cell labelled "Week N" and is not focusable, so the day grid keeps its single tab stop and its arrow keys. See the week_numbers variant.

On the primitive, use_calendar_grid() returns show_week_number() and week_numbers() (one number per row of weeks()), and CalendarGridWeekNumberHeader / CalendarGridWeekNumber render the two cells when you compose your own grid.
