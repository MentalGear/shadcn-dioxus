use dioxus::prelude::*;
use dioxus_primitives::dioxus_attributes::attributes;
use dioxus_primitives::merge_attributes;
use dioxus_primitives::switch;

/// The size of a [`Switch`]. Nova's `data-[size=default]:h-[18.4px] w-[32px]` and
/// `data-[size=sm]:h-[14px] w-[24px]`; the track is not a control-height-ladder
/// step (a switch is not a button-height field), so the sizes are slot variables
/// in `style.css`.
#[derive(Copy, Clone, PartialEq, Default)]
#[non_exhaustive]
pub enum SwitchSize {
    #[default]
    Default,
    Sm,
}

impl SwitchSize {
    pub fn class(&self) -> &'static str {
        match self {
            SwitchSize::Default => "default",
            SwitchSize::Sm => "sm",
        }
    }
}

/// Props for the themed [`Switch`]: the primitive's, plus `size`.
#[derive(Props, Clone, PartialEq)]
pub struct SwitchProps {
    /// The controlled checked state of the switch.
    pub checked: ReadSignal<Option<bool>>,

    /// The default checked state when uncontrolled.
    #[props(default = false)]
    pub default_checked: bool,

    /// Whether the switch is disabled.
    #[props(default = ReadSignal::new(Signal::new(false)))]
    pub disabled: ReadSignal<bool>,

    /// Whether the switch is required in a form.
    #[props(default)]
    pub required: ReadSignal<bool>,

    /// Whether the switch is read-only: focusable and announced (`aria-readonly`), but a click or
    /// Space does not change it, and its value still submits with a form.
    #[props(default)]
    pub read_only: ReadSignal<bool>,

    /// The name attribute for form submission.
    #[props(default)]
    pub name: ReadSignal<String>,

    /// The `id` of the `<form>` the hidden input belongs to (HTML `form` attribute), for a switch outside its form.
    #[props(default)]
    pub form: ReadSignal<Option<String>>,

    /// The value attribute for form submission.
    #[props(default = ReadSignal::new(Signal::new(String::from("on"))))]
    pub value: ReadSignal<String>,

    /// Callback fired when the checked state changes.
    #[props(default)]
    pub on_checked_change: Callback<bool>,

    /// `Default` (32 x 18.4px) or `Sm` (24 x 14px), shadcn's `size` prop.
    #[props(default)]
    pub size: SwitchSize,

    /// Additional attributes to apply to the switch element.
    #[props(extends = GlobalAttributes)]
    pub attributes: Vec<Attribute>,

    /// The children of the switch component.
    pub children: Element,
}

#[component]
pub fn Switch(props: SwitchProps) -> Element {
    let base = attributes!(button {
        class: "dx-switch", "data-dx-root": "",
        "data-size": props.size.class(),
    });
    let merged = merge_attributes(vec![base, props.attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/switch/style.css") }
        switch::Switch {
            checked: props.checked,
            default_checked: props.default_checked,
            disabled: props.disabled,
            required: props.required,
            read_only: props.read_only,
            name: props.name,
            form: props.form,
            value: props.value,
            on_checked_change: props.on_checked_change,
            attributes: merged,
            switch::SwitchThumb { class: "dx-switch-thumb" }
        }
    }
}

// The Switch component includes the switch thumb.
Switch {
    // The state the switch starts in, true for on and false for off. The switch tracks the user's toggles itself;
    // to control it, pass `checked` (a signal) and set it in `on_checked_change`.
    default_checked: true,
    // Callback function triggered when the switch state changes.
    on_checked_change: |checked: bool| {
        // Handle the change in switch state.
    }
}

Component-Specific

What is particular to this component.

size: SwitchSize::Sm shrinks the track to 24 x 14px with a 12px thumb (SwitchSize::Default is 32 x 18.4px, 16px thumb), shadcn's size="sm" | "default". It sets data-size on the switch; the style sheet changes three slot variables (--dx-switch-width, --dx-switch-height, --dx-switch-thumb-size), so the thumb travel follows. See the sizes variant.

Pass form: "form-id".to_string() (the HTML form attribute) and the control's hidden mirror input belongs to that <form> even when it is rendered elsewhere in the page: it submits with it and the form's reset restores it. Leave it out and the nearest ancestor <form> owns it.

read_only: true keeps the switch focusable and announced but unchangeable: a click or Space does nothing, the element carries aria-readonly="true" and a data-readonly presence attribute (absent when the switch is writable), and its value still submits with a form. It differs from disabled, which leaves the tab order, is not announced as editable and submits nothing. A form reset still restores the default. See the read_only variant.

Some props here come in controlled pairs. Controlled and uncontrolled state says how they work, and what freezes a component.

  • SwitchProps: checked with on_checked_change; uncontrolled start value default_checked.
