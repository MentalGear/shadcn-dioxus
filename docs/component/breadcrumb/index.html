Basic

The same sections, in the same order, on every component page.

The Breadcrumb component displays the current page's location within a navigational hierarchy, as a trail of links.

shadcn-dioxus add breadcrumb

Also adds the components and crates this one needs. First time in this app? Set up the CLI and your app first.

Copy the component source and CSS into your app. Import the shared theme CSS once near your app root.

use dioxus::prelude::*;
use dioxus_primitives::AsChild;
use crate::components::icons::{ChevronRight, Ellipsis};
use dioxus_primitives::dioxus_attributes::attributes;
use dioxus_primitives::merge_attributes;

/// A trail of links showing the current page's place in a hierarchy.
/// Renders as `<nav aria-label="breadcrumb"><ol>...</ol></nav>` -- no ARIA
/// widget role beyond the landmark, so there is no APG contract here and no
/// primitive underneath.
#[component]
pub fn Breadcrumb(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    /// The breadcrumb's `BreadcrumbList`.
    children: Element,
) -> Element {
    let base = attributes!(nav { class: "dx-breadcrumb", "data-dx-root": "", "aria-label": "breadcrumb" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/breadcrumb/style.css") }
        nav { ..merged, {children} }
    }
}

/// The ordered list of breadcrumb items.
#[component]
pub fn BreadcrumbList(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    /// The `BreadcrumbItem`s, in order.
    children: Element,
) -> Element {
    let base = attributes!(ol { class: "dx-breadcrumb-list" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/breadcrumb/style.css") }
        ol { ..merged, {children} }
    }
}

/// One `<li>` wrapping a [`BreadcrumbLink`] or [`BreadcrumbPage`].
#[component]
pub fn BreadcrumbItem(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    /// The step to show: a `BreadcrumbLink` or a `BreadcrumbPage`.
    children: Element,
) -> Element {
    let base = attributes!(li { class: "dx-breadcrumb-item" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/breadcrumb/style.css") }
        li { ..merged, {children} }
    }
}

/// A navigable, not-current-page crumb.
#[component]
pub fn BreadcrumbLink(
    #[props(extends = GlobalAttributes)]
    #[props(extends = a)]
    attributes: Vec<Attribute>,
    /// Renders the crumb as the element the callback returns (shadcn's `render` prop: a router
    /// link instead of an `<a>`). The callback gets the crumb's attributes and children
    /// as an [`AsChild`], to spread on and render inside that element.
    r#as: Option<Callback<AsChild, Element>>,
    /// The crumb's text or icon.
    children: Element,
) -> Element {
    let base = attributes!(a { class: "dx-breadcrumb-link" });
    let merged = merge_attributes(vec![base, attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/breadcrumb/style.css") }
        if let Some(dynamic) = r#as {
            {dynamic.call(AsChild::new(merged, children))}
        } else {
            a { ..merged, {children} }
        }
    }
}

/// The current page's crumb -- not a link, marked `aria-current="page"`.
#[component]
pub fn BreadcrumbPage(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    /// The current page's name.
    children: Element,
) -> Element {
    let base = attributes!(span { class: "dx-breadcrumb-page" });
    // `role`/`aria-current`/`aria-disabled` are semantics this wrapper
    // asserts for "the current page" and must win over anything the caller
    // passes -- merged in LAST, after the caller's own attributes. See the
    // merge-precedence policy in dev-docs/issues/duplicate-attribute-findings.md.
    let owned = attributes!(span {
        role: "link",
        "aria-disabled": "true",
        "aria-current": "page",
    });
    let merged = merge_attributes(vec![base, attributes, owned]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/breadcrumb/style.css") }
        span { ..merged, {children} }
    }
}

/// The `>` divider between crumbs. `role="presentation"` +
/// `aria-hidden="true"` per the shadcn/Radix shape -- it carries no
/// semantic content, so it must not be announced.
#[component]
pub fn BreadcrumbSeparator(
    #[props(extends = GlobalAttributes)] attributes: Vec<Attribute>,
    /// A custom divider to show in place of the default `>` chevron.
    #[props(default)] children: Option<Element>,
) -> Element {
    let base = attributes!(li { class: "dx-breadcrumb-separator" });
    let owned = attributes!(li { role: "presentation", "aria-hidden": "true" });
    let merged = merge_attributes(vec![base, attributes, owned]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/breadcrumb/style.css") }
        li {
            ..merged,
            if let Some(children) = &children {
                {children.clone()}
            } else {
                ChevronRight {}
            }
        }
    }
}

/// A collapsed run of crumbs, shown as `...`. A `<span>`, as in shadcn: place
/// it inside a [`BreadcrumbItem`] (`BreadcrumbItem { BreadcrumbEllipsis {} }`).
/// It is not an `<li>`: the usual placement would nest an `<li>` in the
/// item's `<li>`, which is invalid HTML.
///
/// A purely visual marker: `role="presentation"` + `aria-hidden="true"` per the
/// shadcn shape, so the `sr-only` "More" it carries is hidden from assistive
/// technology with the glyph.
#[component]
pub fn BreadcrumbEllipsis(#[props(extends = GlobalAttributes)] attributes: Vec<Attribute>) -> Element {
    let base = attributes!(span { class: "dx-breadcrumb-ellipsis" });
    let owned = attributes!(span { role: "presentation", "aria-hidden": "true" });
    let merged = merge_attributes(vec![base, attributes, owned]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/breadcrumb/style.css") }
        span {
            ..merged,
            Ellipsis {}
            span { "data-dx-visually-hidden": "", "More" }
        }
    }
}

Breadcrumb {
    BreadcrumbList {
        BreadcrumbItem {
            BreadcrumbLink { href: "/", "Home" }
        }
        BreadcrumbSeparator {}
        BreadcrumbItem {
            BreadcrumbLink { href: "/components", "Components" }
        }
        BreadcrumbSeparator {}
        BreadcrumbItem {
            // The current page: not a link, marked `aria-current="page"`.
            BreadcrumbPage { "Breadcrumb" }
        }
    }
}

To collapse a run of crumbs, put BreadcrumbEllipsis {} inside its own BreadcrumbItem. It renders a <span> (not an <li>, so it nests validly in the item) and is hidden from assistive technology, as in shadcn:

BreadcrumbItem {
    BreadcrumbEllipsis {}
}

In a menu trigger

Because it is a span, BreadcrumbEllipsis can be the content of a button, so the collapsed crumbs can open a menu (shadcn's breadcrumb-demo). The ellipsis and its More text are aria-hidden, so name the trigger yourself:

BreadcrumbItem {
    DropdownMenu {
        DropdownMenuTrigger { aria_label: "Toggle menu",
            as: |el: AsChild| rsx! {
                Button { variant: ButtonVariant::Ghost, size: ButtonSize::IconSm, attributes: el.attributes,
                    {el.children}
                }
            },
            BreadcrumbEllipsis {}
        }
        DropdownMenuContent { /* DropdownMenuItem { .. } */ }
    }
}

As a router link

BreadcrumbLink { r#as } renders the crumb as the element the callback returns (shadcn's render prop). The callback gets the crumb's attributes and children as an AsChild; pass the attributes to your router's link and the children inside it, so a click navigates in-app:

BreadcrumbItem {
    BreadcrumbLink {
        r#as: |el: AsChild| rsx! {
            Link { to: Route::Components {}, attributes: el.attributes, {el.children} }
        },
        "Components"
    }
}
