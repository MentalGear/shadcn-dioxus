use dioxus::prelude::*;
use dioxus_primitives::dioxus_attributes::attributes;
use dioxus_primitives::merge_attributes;
use dioxus_primitives::progress::{self, ProgressLabelProps, ProgressProps, ProgressValueProps};

/// A progress bar. The root is the `progressbar` and a flex-wrap row (shadcn's `cn-progress-root`,
/// `flex flex-wrap gap-3`): put a [`ProgressLabel`] and a [`ProgressValue`] in it and they sit on the first
/// line, the label at the start and the value at the end, above the full-width track. With neither, it is the
/// bare 4px track as before.
#[component]
pub fn Progress(props: ProgressProps) -> Element {
    let base = attributes!(div { class: "dx-progress" });
    let merged = merge_attributes(vec![base, props.attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/progress/style.css") }
        progress::Progress {
            value: props.value,
            max: props.max,
            attributes: merged,
            {props.children}
            div { class: "dx-progress-track",
                progress::ProgressIndicator { class: "dx-progress-indicator" }
            }
        }
    }
}

/// The visible name of the [`Progress`] (`text-sm font-medium`); the bar is named by it automatically.
#[component]
pub fn ProgressLabel(props: ProgressLabelProps) -> Element {
    let base = attributes!(span { class: "dx-progress-label" });
    let merged = merge_attributes(vec![base, props.attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/progress/style.css") }
        progress::ProgressLabel { id: props.id, attributes: merged, {props.children} }
    }
}

/// The percent readout of the [`Progress`] (`56%`; `text-muted-foreground`, tabular figures, pushed to the
/// end of the row). Empty while the progress is indeterminate.
#[component]
pub fn ProgressValue(props: ProgressValueProps) -> Element {
    let base = attributes!(span { class: "dx-progress-value" });
    let merged = merge_attributes(vec![base, props.attributes]);
    rsx! {
        document::Link { rel: "stylesheet", href: asset!("/src/components/progress/style.css") }
        progress::ProgressValue { attributes: merged }
    }
}

The Progress component is used to display the progress of a task or operation. It can be used to indicate loading states, file uploads, or any other process that takes time to complete.

Progress {
    // The current progress value (0 to max
    value: 0.5,
    // The maximum value of the progress (default is 100.0)
    max: 1.0,
    // Optional: a ProgressLabel and a ProgressValue, shown above the track
    {children}
}

ProgressLabel is the visible name and ProgressValue the percent readout, as in shadcn. Both go inside Progress: the root is a flex-wrap row, so the label sits at the inline start and the value at the end of the first line, above the full-width track. The bar is named by the label automatically (aria-labelledby, set once the label mounts; your own aria_label/aria_labelledby on Progress wins). ProgressValue renders value / max as a whole-number percent (56%), nothing while the progress is indeterminate, and is aria-hidden because aria-valuenow already carries it.

Progress { value: 56.0,
    ProgressLabel { "Upload progress" }
    ProgressValue {}
}

The track is its own element (.dx-progress-track, 4px, rounded) inside the root, so a Progress without a label or value looks as it did.

Only the indeterminate bar loops: a translateX sweep, 1 s, forever. It is paused whenever the bar is off-screen or the tab is hidden (dioxus_primitives::activity::use_motion_when); a determinate bar subscribes to nothing. The determinate fill animates width for 250 ms, a literal in progress/style.css.

To change them, override animation on .dx-progress-indicator (indeterminate) or its transition (determinate). Neither has a reduced-motion rule yet; add one to your copy if you need it. See Motion for the tokens, reduced motion and the per-component table.
